def read_topic_names():
    names = DEFAULT_TOPIC_NAMES.copy()

    if TOPICS_FILE.exists():
        try:
            with TOPICS_FILE.open(
                "r",
                encoding="utf-8"
            ) as file:
                stored = json.load(file)

            if isinstance(stored, dict):
                names.update({
                    str(k): v
                    for k, v in stored.items()
                    if isinstance(v, str) and v.strip()
                })

        except (OSError, json.JSONDecodeError):
            logger.exception(
                "Cannot read topics.json"
            )

    return names


def save_topic_name(topic_id, name):
    if not name or not name.strip():
        return

    DATA_DIR.mkdir(
        parents=True,
        exist_ok=True
    )

    names = read_topic_names()
    key = str(topic_id)

    if names.get(key) == name:
        return

    names[key] = name

    temporary = TOPICS_FILE.with_suffix(
        ".json.tmp"
    )

    with temporary.open(
        "w",
        encoding="utf-8"
    ) as file:
        json.dump(
            names,
            file,
            ensure_ascii=False,
            indent=2
        )

    temporary.replace(TOPICS_FILE)

    logger.info(
        "TOPIC_NAME_UPDATED topic_id=%s name=%s",
        topic_id,
        name
    )


async def track_topic_changes(
    update: Update,
    context: ContextTypes.DEFAULT_TYPE
):
    message = update.effective_message
    chat = update.effective_chat

    if (
        not message
        or not chat
        or chat.id != CLUB_CHAT_ID
    ):
        return

    if message.forum_topic_created:
        save_topic_name(
            message.message_thread_id,
            message.forum_topic_created.name
        )

    elif (
        message.forum_topic_edited
        and message.forum_topic_edited.name is not None
    ):
        save_topic_name(
            message.message_thread_id,
            message.forum_topic_edited.name
        )

    elif (
        message.general_forum_topic_unhidden
        or message.general_forum_topic_hidden
    ):
        save_topic_name(0, "General")


def read_archive():
    records = []

    if not JSON_FILE.exists():
        return records

    with JSON_FILE.open(
        "r",
        encoding="utf-8"
    ) as file:
        for line in file:
            if not line.strip():
                continue

            try:
                record = json.loads(line)
            except json.JSONDecodeError:
                logger.warning(
                    "Skipping invalid archive line"
                )
                continue

            if record.get("chat_id") == CLUB_CHAT_ID:
                records.append(record)

    return records


def build_archive_context(records):
    parts = []
    total_chars = 0
    topic_names = read_topic_names()

    for record in reversed(records):
        text = str(record.get("text", ""))
        timestamp = str(
            record.get("timestamp", "")
        )
        topic_id = record.get("topic_id", 0)

        topic_name = topic_names.get(
            str(topic_id),
            f"Неизвестная тема (ID: {topic_id})"
        )

        entry = (
            f"Дата: {timestamp}\n"
            f"Тема Telegram: {topic_name} "
            f"(ID: {topic_id})\n"
            f"Сообщение: {text}\n"
        )

        if total_chars + len(entry) > MAX_ARCHIVE_CHARS:
            break

        parts.append(entry)
        total_chars += len(entry)

    parts.reverse()

    return "\n\n".join(parts)


def read_dialog_state():
    if not DIALOG_STATE_FILE.exists():
        return {}

    try:
        with DIALOG_STATE_FILE.open(
            "r",
            encoding="utf-8"
        ) as file:
            state = json.load(file)

        if not isinstance(state, dict):
            raise ValueError(
                "Invalid dialog state format"
            )

        return state

    except (
        OSError,
        json.JSONDecodeError,
        ValueError
    ) as error:
        logger.exception(
            "Cannot read dialog state"
        )
        raise RuntimeError(
            "Не удалось прочитать "
            "состояние диалога."
        ) from error

def get_session_id(state, user_id):
    value = state.get(str(user_id), 0)

    try:
        return int(value)
    except (ValueError, TypeError):
        raise RuntimeError(
            "Некорректный номер сессии диалога."
        )

async def save_message(
    update: Update,
    context: ContextTypes.DEFAULT_TYPE
):
    message = update.effective_message
    chat = update.effective_chat

    if (
        not message
        or not chat
        or chat.id != CLUB_CHAT_ID
        or not message.text
    ):
        return

    timestamp = datetime.now(
        timezone.utc
    ).isoformat()

    topic_id = message.message_thread_id or 0

    sender_id = (
        message.from_user.id
        if message.from_user
        else 0
    )

    record = {
        "timestamp": timestamp,
        "chat_id": chat.id,
        "topic_id": topic_id,
        "user_id": sender_id,
        "message_id": message.id,
        "text": message.text
    }

    DATA_DIR.mkdir(
        parents=True,
        exist_ok=True
    )

    with TEXT_FILE.open(
        "a",
        encoding="utf-8"
    ) as file:
        file.write(
            f"{timestamp} | chat={chat.id} "
            f"| topic={topic_id} "
            f"| user={sender_id} "
            f"| message={message.id}\n"
            f"{message.text}\n\n"
        )

    with JSON_FILE.open(
        "a",
        encoding="utf-8"
    ) as file:
        file.write(
            json.dumps(
                record,
                ensure_ascii=False
            ) + "\n"
        )

    logger.info(
        "GROUP_MESSAGE_SAVED message_id=%s",
        message.id
    )


async def answer_archive_question(
    update: Update,
    question: str
):
    message = update.effective_message

    if not is_owner(update):
        await message.reply_text(
            "Эта функция доступна "
            "только администратору."
        )
        return

    if not question.strip():
        await message.reply_text(
            "Напишите вопрос после "
            "команды /analyze.\n"
            "Например:\n"
            "/analyze Какие темы обсуждались "
            "в клубе?"
        )
        return

    try:
        records = read_archive()
        archive_context = build_archive_context(
            records
        )

        await message.reply_text(
            "Анализирую архив, подождите..."
        )

        answer = await asyncio.to_thread(
            request_gemini,
            question,
            archive_context
        )

        await send_long_message(
            message,
            answer
        )

    except Exception as error:
        logger.error(
            "Archive analysis failed: %s",
            type(error).__name__
        )

        await send_error_response(
            update,
            error,
            "Анализ архива клуба"
        )


async def analyze(
    update: Update,
    context: ContextTypes.DEFAULT_TYPE
):
    question = " ".join(
        context.args or []
    )

    await answer_archive_question(
        update,
        question
    )


def read_recent_dialog(
    user_id,
    session_id
):
    if not DIALOG_HISTORY_FILE.exists():
        return []

    recent_pairs = []

    try:
        with DIALOG_HISTORY_FILE.open(
            "r",
            encoding="utf-8"
        ) as file:
            for line in file:
                if not line.strip():
                    continue

                record = json.loads(line)

                if not isinstance(record, dict):
                    continue

                if record.get("user_id") != user_id:
                    continue

                if record.get("session_id") != session_id:
                    continue

                question = record.get("question")
                answer = record.get("answer")

                if not (
                    isinstance(question, str)
                    and isinstance(answer, str)
                ):
                    continue

                recent_pairs.append(
                    (question, answer)
                )

                if len(recent_pairs) > MAX_DIALOG_PAIRS:
                    recent_pairs.pop(0)

    except (
        OSError,
        json.JSONDecodeError
    ) as error:
        logger.exception(
            "Cannot read private dialog history"
        )
        raise RuntimeError(
            "Не удалось прочитать "
            "историю диалога."
        ) from error

    selected = []
    total_chars = 0

    for question, answer in reversed(recent_pairs):
        pair_size = len(question) + len(answer)

        if total_chars + pair_size > MAX_DIALOG_CHARS:
            break

        selected.append((question, answer))
        total_chars += pair_size

    selected.reverse()

    contents = []

    for question, answer in selected:
        contents.append({
            "role": "user",
            "parts": [{"text": question}]
        })

        contents.append({
            "role": "model",
            "parts": [{"text": answer}]
        })

    return contents
